SELECT
    COUNT(*) AS trades,

    SUM(CASE WHEN net_pnl > 0 THEN 1 ELSE 0 END) AS winners,

    SUM(CASE WHEN net_pnl < 0 THEN 1 ELSE 0 END) AS losers,

    ROUND(
        100.0 *
        SUM(CASE WHEN net_pnl > 0 THEN 1 ELSE 0 END)
        / COUNT(*),
        2
    ) AS win_rate,

    ROUND(SUM(gross_pnl), 2) AS gross_pnl,

    ROUND(SUM(charges), 2) AS charges,

    ROUND(SUM(net_pnl), 2) AS net_pnl

FROM trades

WHERE status='CLOSED'
AND DATE(exit_time)=DATE('now');

3. Exit Reason Breakdown
SELECT
    exit_reason,
    COUNT(*) AS trades,
    ROUND(AVG(net_pnl),2) AS avg_pnl,
    ROUND(SUM(net_pnl),2) AS total_pnl
FROM trades
WHERE status='CLOSED'
AND DATE(exit_time)=DATE('now')
GROUP BY exit_reason;
